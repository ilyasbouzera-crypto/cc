📦 Backend Node.js requis
// npm install googleapis jsonwebtoken const { google } = require('googleapis'); const jwt = require('jsonwebtoken'); const creds = require('./service-account.json'); // POST /api/wallet/google/generate app.post('/api/wallet/google/generate', async (req, res) => { const { customerId, customerName, points, tier } = req.body; const issuerId = process.env.GW_ISSUER_ID; const classSuffix = process.env.GW_CLASS_SUFFIX; // 1. Créer/patcher l'objet de fidélité via l'API const auth = new google.auth.GoogleAuth({ credentials: creds, scopes: ['https://www.googleapis.com/auth/wallet_object.issuer'] }); const client = google.walletobjects({ version: 'v1', auth }); const objectId = `${issuerId}.lp_${customerId}`; try { await client.loyaltyobject.patch({ resourceId: objectId, requestBody: { loyaltyPoints: { balance: { int: points } } } }); } catch(e) { if (e.response?.status === 404) { await client.loyaltyobject.insert({ requestBody: { id: objectId, classId: `${issuerId}.${classSuffix}`, state: 'ACTIVE', accountId: String(customerId), accountName: customerName, loyaltyPoints: { balance: { int: points }, label: 'Points' }, barcode: { type: 'QR_CODE', value: `lp:${customerId}` } }}); } } // 2. Signer le JWT et retourner le lien Save to Wallet const token = jwt.sign({ iss: creds.client_email, aud: 'google', typ: 'savetowallet', iat: Math.floor(Date.now() / 1000), origins: [process.env.ORIGIN_URL], payload: { loyaltyObjects: [{ id: objectId }] } }, creds.private_key, { algorithm: 'RS256' }); res.json({ saveUrl: `https://pay.google.com/gp/v/save/${token}` }); });
